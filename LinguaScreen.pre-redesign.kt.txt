/** One row of a project's timeline. */
private sealed interface TimelineRow {
    val key: Any
    data class Session(val start: Long, val items: List<Caption>) : TimelineRow { override val key get() = "s-$start" }
    data class Gap(val id: Long, val ms: Long) : TimelineRow { override val key get() = "gap-$id" }
    data class Piece(val caption: Caption) : TimelineRow { override val key get() = caption.id }
}

private fun timeline(captions: List<Caption>): List<TimelineRow> = buildList {
    val bySession = captions.groupBy { it.session }
    var prev: Caption? = null
    for (c in captions) {
        if (prev == null || prev.session != c.session) {
            add(TimelineRow.Session(c.session, bySession[c.session].orEmpty()))
        } else {
            // Inside a session the timeline also shows the quiet stretches between pieces of speech.
            val gap = c.time - (prev.time + prev.durationMs)
            if (gap >= SILENCE_GAP_MS) add(TimelineRow.Gap(c.id, gap))
        }
        add(TimelineRow.Piece(c))
        prev = c
    }
}

@Composable
private fun ProjectScreen(project: Project, onToggleListening: () -> Unit) {
    val running by AppState.running.collectAsStateWithLifecycle()
    val captions by AppState.captions.collectAsStateWithLifecycle()
    val playing by PlayAll.current.collectAsStateWithLifecycle()
    val listState = rememberLazyListState()
    val rows = remember(captions) { timeline(captions) }
    var renaming by remember { mutableStateOf(false) }

    // Leaving a project stops listening there and saves it.
    val leave = {
        if (AppState.running.value) onToggleListening()
        PlayAll.stop()
        SegmentAudio.stop()
        ProjectStore.close()
    }
    BackHandler(onBack = leave)

    LaunchedEffect(captions.size) {
        val last = listState.layoutInfo.totalItemsCount - 1
        if (captions.isNotEmpty() && last >= 0 && playing == null) listState.animateScrollToItem(last)
    }
    // Play all: keep the piece being played on screen.
    LaunchedEffect(playing) {
        val i = rows.indexOfFirst { it is TimelineRow.Piece && it.caption.id == playing }
        if (i >= 0) listState.animateScrollToItem(i)
    }

    Column(
        Modifier
            .fillMaxSize()
            .background(MaterialTheme.colorScheme.background)
            .safeDrawingPadding()
            .imePadding()
            .padding(horizontal = 16.dp, vertical = 12.dp)
    ) {
        Row(verticalAlignment = Alignment.CenterVertically) {
            IconButton(onClick = leave) {
                Icon(Icons.AutoMirrored.Filled.ArrowBack, contentDescription = "All projects", tint = MaterialTheme.colorScheme.onSurface)
            }
            Column(Modifier.weight(1f).clickable { renaming = true }) {
                Text(project.name, fontSize = 22.sp, fontWeight = FontWeight.Bold, color = MaterialTheme.colorScheme.onSurface, maxLines = 1)
                Text("Thai ⇄ English · tap the name to rename", color = Muted, fontSize = 13.sp)
            }
            Button(
                onClick = onToggleListening,
                colors = ButtonDefaults.buttonColors(
                    containerColor = if (running) Color(0xFFE5484D) else Accent,
                    contentColor = if (running) Color.White else Color(0xFF00281C),
                ),
            ) {
                Icon(if (running) Icons.Filled.Stop else Icons.Filled.Mic, contentDescription = null)
                Spacer(Modifier.width(8.dp))
                Text(if (running) "Stop" else "Start listening")
            }
        }

        Spacer(Modifier.height(8.dp))
        SpeakToggle()
        Spacer(Modifier.height(8.dp))
        RuntimePanel()
        Spacer(Modifier.height(8.dp))
        PlayAllBar(captions)
        Spacer(Modifier.height(8.dp))

        LazyColumn(
            state = listState,
            modifier = Modifier.weight(1f).fillMaxWidth(),
            verticalArrangement = Arrangement.spacedBy(10.dp),
        ) {
            if (captions.isEmpty()) {
                item {
                    Text(
                        if (running) "Listening… speak Thai or English nearby."
                        else "Tap Start listening. Everything you record here is saved in this project.",
                        color = Muted, modifier = Modifier.padding(vertical = 24.dp),
                    )
                }
            }
            items(rows, key = { it.key }) { row ->
                when (row) {
                    is TimelineRow.Session -> SessionHeader(row)
                    is TimelineRow.Gap -> SilenceGap(row.ms)
                    is TimelineRow.Piece -> CaptionCard(row.caption, highlighted = row.caption.id == playing)
                }
            }
        }

        Spacer(Modifier.height(8.dp))
        PushToTalkButton(running)
        Spacer(Modifier.height(8.dp))
        TypeToTranslate()
    }

    if (renaming) NameDialog("Rename project", project.name, onDismiss = { renaming = false }) {
        ProjectStore.rename(project, it)
        renaming = false
    }
}

private val dayFormat = SimpleDateFormat("EEE d MMM · HH:mm", Locale.getDefault())

private fun count(n: Int, word: String) = if (n == 1) "1 $word" else "$n ${word}s"

private fun duration(ms: Long): String {
    val s = ms / 1000
    return when {
        s < 60 -> "$s s"
        s < 3600 -> "${s / 60} min"
        else -> "${s / 3600} h ${(s % 3600) / 60} min"
    }
}

@Composable
private fun SessionHeader(row: TimelineRow.Session) {
    val end = row.items.maxOfOrNull { it.time + it.durationMs } ?: row.start
    Row(verticalAlignment = Alignment.CenterVertically, modifier = Modifier.fillMaxWidth().padding(top = 8.dp)) {
        Column(Modifier.weight(1f)) {
            Text(dayFormat.format(Date(row.start)), color = MaterialTheme.colorScheme.onSurface, fontSize = 14.sp, fontWeight = FontWeight.SemiBold)
            Text("Session · ${duration(end - row.start)} · ${count(row.items.size, "piece")}", color = Muted, fontSize = 12.sp)
        }
        TextButton(onClick = { PlayAll.start(row.items) }) {
            Icon(Icons.Filled.PlayArrow, contentDescription = null, modifier = Modifier.size(18.dp))
            Spacer(Modifier.width(4.dp))
            Text("Play session")
        }
    }
}

/** Plays every recording of the project in order: original voices, or the translations spoken. */
@Composable
private fun PlayAllBar(captions: List<Caption>) {
    val playing by PlayAll.current.collectAsStateWithLifecycle()
    val mode by PlayAll.mode.collectAsStateWithLifecycle()
    val fast by PlayAll.fast.collectAsStateWithLifecycle()
    val position by PlayAll.position.collectAsStateWithLifecycle()
    val count = captions.count { it.done && (it.heard.isNotBlank() || it.turns.isNotEmpty()) }
    if (count == 0) return

    Row(
        verticalAlignment = Alignment.CenterVertically,
        modifier = Modifier.fillMaxWidth().clip(RoundedCornerShape(16.dp)).background(Surface).padding(horizontal = 6.dp, vertical = 2.dp),
    ) {
        if (playing == null) {
            TextButton(onClick = { PlayAll.start(captions) }) {
                Icon(Icons.Filled.PlayArrow, contentDescription = null)
                Spacer(Modifier.width(4.dp))
                Text("Play all · $count")
            }
        } else {
            IconButton(onClick = { PlayAll.stop() }) { Icon(Icons.Filled.Stop, contentDescription = "Stop", tint = Accent) }
            IconButton(onClick = { PlayAll.previous() }) {
                Icon(Icons.Filled.SkipPrevious, contentDescription = "Previous", tint = MaterialTheme.colorScheme.onSurface)
            }
            Text("${position.first} / ${position.second}", color = MaterialTheme.colorScheme.onSurface, fontSize = 13.sp)
            IconButton(onClick = { PlayAll.next() }) {
                Icon(Icons.Filled.SkipNext, contentDescription = "Next", tint = MaterialTheme.colorScheme.onSurface)
            }
        }
        Spacer(Modifier.weight(1f))
        ToggleChip("Original voices", mode == PlayMode.ORIGINAL) { PlayAll.setMode(PlayMode.ORIGINAL) }
        Spacer(Modifier.width(6.dp))
        ToggleChip("Translations spoken", mode == PlayMode.TRANSLATION) { PlayAll.setMode(PlayMode.TRANSLATION) }
        Spacer(Modifier.width(6.dp))
        ToggleChip(if (fast) "1.5×" else "1×", fast) { PlayAll.toggleSpeed() }
        Spacer(Modifier.width(6.dp))
    }
}

private val Amber = Color(0xFFF5A524)
private val Blue = Color(0xFF5B9DFF)

/** Last ~5 s of mic audio. Green = speech being recorded, amber = mic paused for our own audio. */
@Composable
private fun Waveform() {
    val bars by AppState.waveform.collectAsStateWithLifecycle()
    Row(verticalAlignment = Alignment.CenterVertically) {
        Text("Mic", color = Muted, fontSize = 12.sp, modifier = Modifier.width(130.dp))
        Canvas(Modifier.weight(1f).height(44.dp)) {
            val n = 150
            val step = size.width / n
            val barW = (step * 0.6f).coerceAtLeast(1f)
            val mid = size.height / 2
            val offset = n - bars.size
            bars.forEachIndexed { i, b ->
                val h = (b.level.coerceAtLeast(0.03f) * size.height).coerceAtMost(size.height)
                val x = (offset + i) * step
                drawRoundRect(
                    color = when {
                        b.muted -> Amber
                        b.speech -> Accent
                        else -> Idle
                    },
                    topLeft = Offset(x, mid - h / 2),
                    size = Size(barW, h),
                    cornerRadius = CornerRadius(barW / 2),
                )
            }
        }
    }
}

/** Where every heard segment is right now, so nothing disappears silently. */
@Composable
private fun QueueStrip() {
    val captions by AppState.captions.collectAsStateWithLifecycle()
    val buffered by AppState.bufferedBytes.collectAsStateWithLifecycle()
    val gain by AppState.inputGain.collectAsStateWithLifecycle()
    val live by AppState.liveStatus.collectAsStateWithLifecycle()
    val recording = captions.count { it.status == SegmentStatus.RECORDING }
    val queued = captions.count { it.status == SegmentStatus.QUEUED }
    val translating = captions.count { it.status == SegmentStatus.TRANSLATING }
    val done = captions.count { it.status == SegmentStatus.DONE }
    val bufferedSec = buffered / (2f * Config.MIC_SAMPLE_RATE)
    val offline = !live.startsWith("Live")

    Column(verticalArrangement = Arrangement.spacedBy(4.dp)) {
        Row(verticalAlignment = Alignment.CenterVertically, horizontalArrangement = Arrangement.spacedBy(6.dp)) {
            Text("Queue", color = Muted, fontSize = 12.sp, modifier = Modifier.width(124.dp))
            Chip("● Recording $recording", if (recording > 0) Color(0xFFE5484D) else Idle)
            Chip("Queued $queued", if (queued > 0) Amber else Idle)
            Chip("Translating $translating", if (translating > 0) Blue else Idle)
            Chip("Done $done", if (done > 0) Accent else Idle)
            Chip("Boost ×%.1f".format(gain), if (gain > 1.3f) Accent else Idle)
        }
        if (bufferedSec >= 1f || (offline && queued + translating > 0)) {
            Text(
                "⚠ %.0f s of speech is saved and waiting · it will be sent as soon as Gemini reconnects".format(bufferedSec),
                color = Amber, fontSize = 12.sp,
            )
        }
    }
}

@Composable
private fun Chip(text: String, color: Color) {
    Text(
        text, color = color, fontSize = 11.sp, fontWeight = FontWeight.Medium,
        modifier = Modifier.clip(RoundedCornerShape(8.dp)).background(color.copy(alpha = 0.12f))
            .padding(horizontal = 8.dp, vertical = 3.dp),
    )
}

@Composable

        Text(label, color = Muted, fontSize = 12.sp, modifier = Modifier.width(130.dp))

            progress = { value.coerceIn(0f, 1f) },
            modifier = Modifier.weight(1f).height(6.dp).clip(RoundedCornerShape(3.dp)),
            color = Accent, trackColor = Color(0xFF2A303A),
        )
    }
}

@Composable

private val speakerColors = listOf(
    Color(0xFF4CC38A), Color(0xFF52A9FF), Color(0xFFF5A524), Color(0xFFE879F9),
    Color(0xFFFF6B6B), Color(0xFF2DD4BF), Color(0xFFA3E635), Color(0xFFFB923C),
)

private fun speakerColor(n: Int) = speakerColors[(n - 1).mod(speakerColors.size)]

@Composable
private fun CaptionCard(caption: Caption, highlighted: Boolean = false) {
    val scope = rememberCoroutineScope()
    var loading by remember { mutableStateOf(false) }

    Card(
        colors = CardDefaults.cardColors(containerColor = if (caption.mine) Color(0xFF16202E) else Surface),
        shape = RoundedCornerShape(16.dp),
        border = if (highlighted) BorderStroke(2.dp, Accent) else null,
        modifier = Modifier.fillMaxWidth(),
    ) {
        Column(Modifier.padding(start = 14.dp, end = 6.dp, top = 6.dp, bottom = 12.dp)) {
            val replaying by AppState.replaying.collectAsStateWithLifecycle()
            val (statusText, statusColor) = when (caption.status) {
                SegmentStatus.RECORDING -> "● Recording" to Color(0xFFE5484D)
                SegmentStatus.QUEUED -> "Queued" to Amber
                SegmentStatus.TRANSLATING -> "Translating…" to Blue
                SegmentStatus.DONE -> "✓ Done" to Accent
                SegmentStatus.NO_SPEECH -> "Silence · no words" to Muted
                SegmentStatus.SKIPPED -> (caption.note ?: "Noise · skipped on device") to Muted
            }
            Row(verticalAlignment = Alignment.CenterVertically) {
                Text(
                    timeFormat.format(Date(caption.time)) + " · %.1f s".format(caption.durationMs / 1000f),
                    color = MaterialTheme.colorScheme.onSurface, fontSize = 12.sp, fontWeight = FontWeight.Medium,
                )
                Spacer(Modifier.width(8.dp))
                Chip(statusText, statusColor)
                if (caption.mine) {
                    Spacer(Modifier.width(6.dp))
                    Chip("You · English → Thai", Blue)
                }
                if (caption.offline) {
                    Spacer(Modifier.width(6.dp))
                    Chip("Offline", Amber)
                }
                caption.speaker?.let {
                    Spacer(Modifier.width(6.dp))
                    Chip("Speaker $it", speakerColor(it))
                }
                Spacer(Modifier.weight(1f))
                caption.audioFile?.let { path ->
                    IconButton(onClick = { if (replaying) SegmentAudio.stop() else SegmentAudio.play(path, caption.heard) }) {
                        Icon(
                            if (replaying) Icons.Filled.Stop else Icons.Filled.PlayArrow,
                            contentDescription = "Play original audio", tint = Muted,
                        )
                    }
                } ?: Spacer(Modifier.height(48.dp))
            }
            caption.note?.takeIf { !caption.done || caption.status == SegmentStatus.NO_SPEECH }?.let {
                Text(it, color = Amber, fontSize = 13.sp, modifier = Modifier.padding(bottom = 4.dp))
            }
            if (caption.turns.isNotEmpty()) {
                // More than one voice inside this piece: show who said what (A, B are per piece).
                caption.turns.forEach { t ->
                    Row(Modifier.padding(end = 8.dp, top = 4.dp)) {
                        Text("${t.speaker}  ", color = Accent, fontSize = 16.sp, fontWeight = FontWeight.Bold)
                        Column {
                            Text(t.heard, color = Color(0xFFC9D1DC), fontSize = 16.sp)
                            if (t.translation.isNotBlank()) {
                                Text(t.translation, color = MaterialTheme.colorScheme.onSurface, fontSize = 18.sp, fontWeight = FontWeight.Medium)
                            }
                        }
                    }
                }
            } else if (caption.heard.isNotBlank()) {
                Text(caption.heard.trim(), color = Color(0xFFC9D1DC), fontSize = 18.sp, modifier = Modifier.padding(end = 8.dp))
            } else if (caption.preview.isNotBlank() && !caption.done) {
                Text(caption.preview.trim(), color = Muted, fontSize = 18.sp, modifier = Modifier.padding(end = 8.dp))
            } else if (caption.status == SegmentStatus.TRANSLATING) {
                Text("Listening to the recording…", color = Muted, fontSize = 14.sp)
            } else if (caption.status == SegmentStatus.NO_SPEECH || caption.status == SegmentStatus.SKIPPED) {
                Text("(silence / no words)", color = Muted, fontSize = 14.sp)
            }
            if (caption.translation.isNotBlank() && caption.turns.isEmpty()) {
                Spacer(Modifier.height(4.dp))
                Text(caption.translation.trim(), color = MaterialTheme.colorScheme.onSurface, fontSize = 20.sp, fontWeight = FontWeight.Medium)
            }
            caption.explanation?.let {
                Spacer(Modifier.height(8.dp))
                Text(it, color = Color(0xFFC9D1DC), fontSize = 14.sp)
                Text(caption.explainSource.orEmpty(), color = Accent, fontSize = 11.sp)
            }
            if (caption.done && caption.explanation == null && caption.translation.isNotBlank()) {
                TextButton(
                    enabled = !loading,
                    onClick = {
                        loading = true
                        scope.launch {
                            val answer = runCatching { HybridAssistant.explain(caption.heard, caption.translation) }
                            AppState.updateCaption(caption.id) {
                                answer.fold(
                                    { a -> it.copy(explanation = a.text, explainSource = a.source) },
                                    { e -> it.copy(explanation = "Couldn't explain: ${e.message?.take(100)}", explainSource = "") },
                                )
                            }
                            loading = false
                        }
                    },
                ) { Text(if (loading) "Thinking…" else "What did they mean?") }
            }
        }
    }
}

@Composable
private fun TypeToTranslate() {
    val scope = rememberCoroutineScope()
    var text by remember { mutableStateOf("") }
    var result by remember { mutableStateOf("") }

    Column {
        if (result.isNotBlank()) {
            Text(
                result, color = MaterialTheme.colorScheme.onSurface, fontSize = 16.sp,
                modifier = Modifier.fillMaxWidth().clip(RoundedCornerShape(12.dp)).background(Surface)
                    .clickable { result = "" }.padding(12.dp),
            )
            Spacer(Modifier.height(6.dp))
        }
        Row(verticalAlignment = Alignment.CenterVertically) {
            OutlinedTextField(
                value = text, onValueChange = { text = it },
                placeholder = { Text("Type Thai or English (works offline)") },
                singleLine = true, modifier = Modifier.weight(1f),
            )
            Spacer(Modifier.width(8.dp))
            Button(
                enabled = text.isNotBlank(),
                onClick = {
                    val input = text
                    scope.launch {
                        result = runCatching { OfflineTranslator.translate(input) }
                            .getOrElse { "Offline translation failed: ${it.message}" }
                    }
                },
            ) { Text("Translate") }
        }
    }
}
